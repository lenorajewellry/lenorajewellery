function toggleMenu(){

  document
    .getElementById("menu")
    .classList.toggle("open");

}


function hideAll(){

  document
    .querySelectorAll(".page")
    .forEach(page=>{

      page.classList.remove("active");

    });

}


function showPage(id){

  hideAll();

  const page =
    document.getElementById(id);

  if(page){

    page.classList.add("active");

  }

  document
    .getElementById("menu")
    .classList.remove("open");

  window.scrollTo({
    top:0,
    behavior:"smooth"
  });

}


async function showCategory(category){

  hideAll();

  const page =
    document.getElementById("category-page");

  page.classList.add("active");

  document
    .getElementById("menu")
    .classList.remove("open");

  const info =
    categories[category];

  document
    .getElementById("category-title")
    .textContent =
    info.title;

  document
    .getElementById("category-description")
    .textContent =
    info.description;

  await loadProducts(category);

  window.scrollTo({
    top:0,
    behavior:"smooth"
  });

}


async function loadProducts(category){

  const container =
    document.getElementById("products");

  container.innerHTML =
    '<div class="loading">Loading...</div>';


  const {
    data,
    error
  } =
    await db
      .from("products")
      .select("*")
      .eq("category",category)
      .order(
        "sort_order",
        {ascending:true}
      );


  if(error){

    container.innerHTML =
      '<div class="loading">Unable to load products.</div>';

    console.error(error);

    return;

  }


  if(!data || data.length === 0){

    container.innerHTML =
      '<div class="loading">No products yet.</div>';

    return;

  }


  container.innerHTML = "";


  data.forEach(product=>{

    const card =
      document.createElement("a");


    card.className =
      "product";


    card.href =
      product.link ||
      INSTAGRAM_URL;


    card.target =
      "_blank";


    card.rel =
      "noopener";


    let imageHTML = "";


    if(product.image_url){

      imageHTML =
        `<img
          src="${escapeHTML(product.image_url)}"
          alt="${escapeHTML(product.name)}">`;

    }else{

      imageHTML = `

        <div class="no-image">

          <span>✦</span>

          <small>
            Coming Soon
          </small>

        </div>

      `;

    }


    card.innerHTML = `

      <div class="product-image">

        ${imageHTML}

      </div>

      <div class="product-name">

        ${escapeHTML(product.name)}

      </div>

    `;


    container.appendChild(card);

  });

}


function escapeHTML(value){

  return String(value ?? "")

    .replace(/&/g,"&amp;")

    .replace(/</g,"&lt;")

    .replace(/>/g,"&gt;")

    .replace(/"/g,"&quot;")

    .replace(/'/g,"&#039;");

}


async function loadSettings(){

  const {
    data,
    error
  } =
    await db
      .from("site_settings")
      .select("*")
      .eq("key","instagram")
      .maybeSingle();


  const link =
    document.getElementById(
      "instagram-link"
    );


  if(!error && data && data.value){

    link.href =
      data.value;

  }else{

    link.href =
      INSTAGRAM_URL;

  }


  link.textContent =
    "Instagram · " +
    INSTAGRAM_USERNAME;

}


loadSettings();
